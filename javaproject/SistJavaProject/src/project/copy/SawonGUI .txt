package project;

import javax.swing.*;
import javax.swing.table.*;
import java.awt.*;
import java.awt.event.*;
import java.util.*;
import java.util.List;

public class SawonGUI extends JFrame {

	JTable table;
	private DefaultTableModel tableModel;
	private SawonController controller;

	public SawonGUI() {
		controller = new SawonController(this);
		setTitle("사원 관리");
		setBounds(500, 200, 800, 500);
		setDefaultCloseOperation(EXIT_ON_CLOSE);
		setLayout(new BorderLayout());

		String[] columnNames = { "사원번호", "사원이름", "성별", "부서", "급여", "입사일" };
		tableModel = new DefaultTableModel(null, columnNames);
		table = new JTable(tableModel);
		JScrollPane scrollPane = new JScrollPane(table);
		add(scrollPane, BorderLayout.CENTER);

		JPanel buttonPanel = createButtonPanel();
		add(buttonPanel, BorderLayout.SOUTH);
	}

	private JPanel createButtonPanel() {
		JPanel buttonPanel = new JPanel(new GridLayout(3, 2, 10, 10));
		buttonPanel.setBorder(BorderFactory.createEmptyBorder(10, 20, 10, 20));

		buttonPanel.add(createButton("회원가입", e -> controller.showInputDialog("회원가입")));
		buttonPanel.add(createButton("전체회원출력", e -> controller.selectAll()));
		buttonPanel.add(createButton("회원정보수정", e -> controller.showUpdateDialog()));
		buttonPanel.add(createButton("회원삭제", e -> controller.deleteMember()));
		buttonPanel.add(createButton("검색", e -> controller.showSearchDialog()));
		buttonPanel.add(createButton("종료", e -> controller.exit()));

		return buttonPanel;
	}

	private JButton createButton(String text, ActionListener action) {
		JButton button = new JButton(text);
		button.addActionListener(action);
		return button;
	}

	public void displayMembers(List<SawonMember> members) {
		tableModel.setRowCount(0);
		for (SawonMember member : members) {
			tableModel.addRow(new Object[] { member.getNum(), member.getName(), member.getGender(), member.getBuseo(),
					member.getPay(), member.getHireday() });
		}
	}

	public void showMessage(String message) {
		JOptionPane.showMessageDialog(this, message);
	}

}
